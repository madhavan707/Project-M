function doPost(e) {
  try {
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    
    if (!e || !e.postData || !e.postData.contents) {
      return ContentService.createTextOutput(JSON.stringify({ 
        status: "error", 
        message: "No POST data received. This function triggers via HTTP POST." 
      })).setMimeType(ContentService.MimeType.JSON);
    }

    var payload = JSON.parse(e.postData.contents);
    var records = Array.isArray(payload) ? payload : (payload.records || [payload]);

    // 15 Capitalized Headers without TIMESTAMP
    if (sheet.getLastRow() === 0) {
      sheet.appendRow([
        "AADHAAR NUMBER",
        "APPLICANT FIRST NAME",
        "APPLICANT MIDDLE NAME",
        "APPLICANT LAST NAME",
        "NAME AS PER AADHAAR",
        "FATHER FIRST NAME",
        "FATHER MIDDLE NAME",
        "FATHER LAST NAME",
        "DOB",
        "GENDER",
        "MOBILE NUMBER",
        "ADDRESS LINE 1",
        "ADDRESS LINE 2",
        "PINCODE",
        "STATE"
      ]);
    }

    // Append all records in CAPITAL CASE
    records.forEach(function(item) {
      sheet.appendRow([
        String(item.aadhar_number || "").toUpperCase(),
        String(item.applicant_first_name || "").toUpperCase(),
        String(item.applicant_middle_name || "").toUpperCase(),
        String(item.applicant_last_name || "").toUpperCase(),
        String(item.name_as_per_aadhar || item.full_name || "").toUpperCase(),
        String(item.father_first_name || "").toUpperCase(),
        String(item.father_middle_name || "").toUpperCase(),
        String(item.father_last_name || "").toUpperCase(),
        String(item.dob || "").toUpperCase(),
        String(item.gender || "").toUpperCase(),
        String(item.mobile_number || "").toUpperCase(),
        String(item.address_line_1 || "").toUpperCase(),
        String(item.address_line_2 || "").toUpperCase(),
        String(item.pincode || "").toUpperCase(),
        String(item.state || "").toUpperCase()
      ]);
    });

    return ContentService.createTextOutput(JSON.stringify({ 
      status: "success", 
      count: records.length 
    })).setMimeType(ContentService.MimeType.JSON);

  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ 
      status: "error", 
      error: err.toString() 
    })).setMimeType(ContentService.MimeType.JSON);
  }
}

Phase 2: One-Click Portal Autofill

Use this free Tampermonkey / Chrome Userscript on your target PAN or government portal. It fetches the currently extracted record dynamically from your local tool and automatically fills all inputs in CAPITAL CASE! 

// ==UserScript==
// @name         Aadhaar to PAN Auto-Filler (Step-by-Step + 1-Click)
// @namespace    http://tampermonkey.net/
// @version      2.0
// @description  Autofills PAN form fields dynamically in CAPITAL CASE with step-by-step chat/button control
// @match        *://*/*
// @grant        GM_xmlhttpRequest
// ==/UserScript==

(function() {
    'use strict';
    let cachedData = null;
    let currentStep = 0;
    const steps = ['applicant', 'parents', 'contact', 'address', 'proofs'];

    // Create Floating Assistant Widget
    const panel = document.createElement('div');
    panel.id = 'pan-autofill-panel';
    panel.style = 'position:fixed;bottom:24px;right:24px;z-index:999999;background:rgba(15,23,42,0.95);border:1px solid rgba(99,102,241,0.4);border-radius:12px;padding:14px;box-shadow:0 10px 30px rgba(0,0,0,0.5);display:flex;flex-direction:column;gap:8px;font-family:sans-serif;width:240px;backdrop-filter:blur(10px);color:#fff;';

    panel.innerHTML = `
      

        ⚡ PAN AUTO-FILLER
        Ready
      

      

        
        
        
        
      

      
      
Press Alt + N for next section

    `;
    document.body.appendChild(panel);

    const logStatus = (msg) => {
        const s = document.getElementById('pan-status-text');
        if (s) s.innerText = msg;
    };

    const fetchData = (cb) => {
        if (cachedData) { cb(cachedData); return; }
        logStatus('Fetching data...');
        GM_xmlhttpRequest({
            method: 'GET',
            url: 'http://127.0.0.1:8000/api/latest-record',
            onload: function(res) {
                try {
                    cachedData = JSON.parse(res.responseText);
                    logStatus('Data Loaded');
                    cb(cachedData);
                } catch(e) { logStatus('Fetch Error'); }
            },
            onerror: function() { logStatus('Offline (8000)'); }
        });
    };

    const setText = (selectors, val) => {
        if (!val) return;
        for (const s of selectors) {
            const el = document.querySelector(s);
            if (el) {
                el.focus();
                el.value = String(val).toUpperCase();
                el.dispatchEvent(new Event('input', { bubbles: true }));
                el.dispatchEvent(new Event('change', { bubbles: true }));
                el.dispatchEvent(new Event('blur', { bubbles: true }));
                return true;
            }
        }
        return false;
    };

    const setSelect = (selectors, matchText) => {
        if (!matchText) return;
        const search = String(matchText).toUpperCase().trim();
        for (const s of selectors) {
            const sel = document.querySelector(s);
            if (sel && sel.tagName === 'SELECT') {
                for (let i = 0; i < sel.options.length; i++) {
                    const opt = sel.options[i];
                    if (opt.text.toUpperCase().includes(search) || opt.value.toUpperCase().includes(search)) {
                        sel.selectedIndex = i;
                        sel.dispatchEvent(new Event('change', { bubbles: true }));
                        return true;
                    }
                }
            }
        }
        return false;
    };

    const setRadio = (matchTexts) => {
        const radios = Array.from(document.querySelectorAll('input[type="radio"]'));
        for (const r of radios) {
            const parentText = (r.parentElement ? r.parentElement.innerText : '').toUpperCase();
            const val = (r.value || '').toUpperCase();
            if (matchTexts.some(t => parentText.includes(t.toUpperCase()) || val.includes(t.toUpperCase()))) {
                r.click();
                r.dispatchEvent(new Event('change', { bubbles: true }));
                return true;
            }
        }
        return false;
    };

    const fillSection = (section, d) => {
        if (section === 'applicant' || section === 'all') {
            setSelect(['select[name*="category" i]', '#category'], 'Individual');
            setText(['input[name*="first" i]', 'input[placeholder*="First Name" i]'], d.applicant_first_name);
            setText(['input[name*="middle" i]', 'input[placeholder*="Middle Name" i]'], d.applicant_middle_name);
            setText(['input[name*="last" i]', 'input[name*="surname" i]', 'input[placeholder*="Last Name" i]'], d.applicant_last_name);
            setText(['input[name*="nameaspaadhar" i]', 'input[name*="aadhaarname" i]'], d.name_as_per_aadhar || d.full_name);
            logStatus('Applicant Filled');
        }
        if (section === 'parents' || section === 'all') {
            setRadio(['No']);
            setRadio(['Father']);
            setText(['input[name*="fatfirst" i]', 'input[name*="fatherfirst" i]'], d.father_first_name);
            setText(['input[name*="fatmiddle" i]', 'input[name*="fathermiddle" i]'], d.father_middle_name);
            setText(['input[name*="fatlast" i]', 'input[name*="fatherlast" i]', 'input[name*="fathersurname" i]'], d.father_last_name);
            logStatus('Father Filled');
        }
        if (section === 'contact' || section === 'all') {
            setText(['input[name*="dob" i]', 'input[placeholder*="dd/MM/yyyy" i]'], d.dob);
            setSelect(['select[name*="resident" i]'], 'Resident');
            setSelect(['select[name*="isd" i]', 'select[name*="country" i]'], '91');
            setText(['input[name*="mobile" i]', 'input[placeholder*="Mobile" i]'], d.mobile_number);
            setSelect(['select[name*="communication" i]'], 'Residence');
            setSelect(['select[name*="state" i]', 'select[name*="dispatch" i]'], d.state || 'TAMIL NADU');
            setText(['input[name*="pin" i]', 'input[placeholder*="PIN" i]'], d.pincode);
            logStatus('Contact Filled');
        }
        if (section === 'proofs' || section === 'all') {
            setSelect(['select[name*="aadhaar" i]'], 'Aadhaar');
            const cleanUid = (d.aadhar_number || '').replace(/\\s+/g, '');
            setText(['input[name*="aadhaar" i]', 'input[name*="uid" i]'], cleanUid);
            setSelect(['select[name*="identity" i]', 'select[name*="poi" i]'], 'Aadhaar');
            setSelect(['select[name*="address" i]', 'select[name*="poa" i]'], 'Aadhaar');
            setSelect(['select[name*="dobproof" i]', 'select[name*="pod" i]'], 'Aadhaar');
            logStatus('Proofs Filled');
        }
    };

    document.getElementById('btn-fill-app').onclick = () => fetchData(d => fillSection('applicant', d));
    document.getElementById('btn-fill-fat').onclick = () => fetchData(d => fillSection('parents', d));
    document.getElementById('btn-fill-con').onclick = () => fetchData(d => fillSection('contact', d));
    document.getElementById('btn-fill-prf').onclick = () => fetchData(d => fillSection('proofs', d));
    document.getElementById('btn-fill-all').onclick = () => fetchData(d => fillSection('all', d));

    window.addEventListener('keydown', (e) => {
        if (e.altKey && (e.key === 'n' || e.key === 'N')) {
            e.preventDefault();
            fetchData(d => {
                fillSection(steps[currentStep % steps.length], d);
                currentStep++;
            });
        }
    });
})();
